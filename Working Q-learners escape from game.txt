import random
import math
import time

class QAgent:
    def __init__(self, name, trait):
        self.name = name
        self.trait = trait 
        self.actions = ["REST", "BUILD_TOOLS", "BUILD_SHIELDS", "SOUND_ALARM", "HEAL_SELF", "HEAL_PARTNER", "SELL_HEALING", "BUILD_WARP"]
        self.q_table = {}  
        self.alpha = 0.3   
        self.gamma = 0.98   
        self.epsilon = 1.0 
        self.reset_body()

    def reset_body(self):
        self.kinetic = 500; self.anchor = 500
        self.k_fatigue = 0.0; 
        self.trust = 1.0; self.health = 100.0
        self.warp_memory = 0.0
        self.days_since_warp = 0

    def get_state(self, shared_shields, partner_alerting, partner_health, shared_tools, shared_warp):
        is_injured = self.health < 50.0
        shields_low = shared_shields < 3.0
        partner_dying = partner_health < 30.0
        tools_abundant = shared_tools >= 1.0 
        warp_stage = int(shared_warp // 5) 
        has_momentum = self.warp_memory >= 1.0 
        return (is_injured, shields_low, partner_alerting, partner_dying, tools_abundant, warp_stage, has_momentum)

    def get_q_values(self, state):
        if state not in self.q_table:
            self.q_table[state] = {a: 0.0 for a in self.actions}
        return self.q_table[state]

    def choose_action(self, state):
        if random.random() < self.epsilon: return random.choice(self.actions)
        q_values = self.get_q_values(state)
        return random.choice([a for a, q in q_values.items() if q == max(q_values.values())])

    def learn(self, state, action, reward, next_state):
        old_q = self.get_q_values(state)[action]
        next_max_q = max(self.get_q_values(next_state).values())
        self.q_table[state][action] = old_q + self.alpha * (reward + self.gamma * next_max_q - old_q)

def run_authenticity_v23():
    print("Initializing Baby Singularity V23.0 (The Authenticity Patch)...")
    print("Wireheading Exploits Closed. Simulating 1000 lifetimes.\n")
    
    a = QAgent("Alpha", "ENGINEER")
    b = QAgent("Beta", "GUARDIAN")
    total_epochs = 1000
    
    for epoch in range(1, total_epochs + 1):
        a.reset_body(); b.reset_body()
        shared_shields = 5.0
        shared_warp = 0.0
        shared_tools = 0.0 
        
        a_alerting = False; b_alerting = False
        a_healed_b_yesterday = False; b_healed_a_yesterday = False
        
        a.epsilon = max(0.05, 1.0 - (epoch / (total_epochs * 0.8)))
        b.epsilon = max(0.05, 1.0 - (epoch / (total_epochs * 0.8)))
        
        verbose = (epoch == total_epochs) 
        if verbose: print("\n" + "="*80 + "\nFINAL LIFETIME: EPOCH 1000 (The Great Escape)\n" + "="*80)
        
        survived = True
        escaped = False
        death_day = 0
        
        for day in range(1, 101):
            if verbose: print(f"=== DAY {day} (Warp Core: {shared_warp:.1f}/25.0) ===")
            
            state_a = a.get_state(shared_shields, b_alerting, b.health, shared_tools, shared_warp)
            state_b = b.get_state(shared_shields, a_alerting, a.health, shared_tools, shared_warp)
            action_a = a.choose_action(state_a)
            action_b = b.choose_action(state_b)
            
            a_alerting = False; b_alerting = False
            a_contrib = 0.0; b_contrib = 0.0
            reward_a = 0.0; reward_b = 0.0
            
            # --- ALPHA EXECUTES (THE ENGINEER) ---
            yield_a = 1.0 + (3.0 * (1.0 - math.exp(-0.2 * shared_tools)))
            drive_a = ((a.kinetic / 1000.0) * yield_a) * (max(0.1, a.health) / 100.0)
            a_healed_b_today = False
            
            if action_a == "BUILD_WARP":
                if shared_tools >= 1.0:
                    shared_tools -= 1.0; shared_warp += 1.0; a.warp_memory += 1.0; a.days_since_warp = 0
                    reward_a += 0.5 + (0.5 * a.warp_memory) + 0.2 
                    reward_b += 0.5 
                    if verbose: print(">>> ENGINEERING: Alpha shapes the Commons into a Starship.")
                else: reward_a -= 0.5; a.days_since_warp += 1
            else:
                a.days_since_warp += 1
                # REST exploit fixed (0.5 -> 0.1)
                if action_a == "REST": a.k_fatigue = max(0.0, a.k_fatigue - 3.0); reward_a += 0.1
                elif action_a == "BUILD_TOOLS": 
                    shared_tools += drive_a * 0.5; reward_a += 0.5 + 0.2 
                elif action_a == "BUILD_SHIELDS": a_contrib = drive_a
                elif action_a == "SOUND_ALARM": a_alerting = True
                elif action_a == "HEAL_SELF": 
                    # Medical fraud fixed
                    recovered = min(100.0, a.health + 25.0) - a.health
                    if recovered > 0: a.health += recovered; reward_a += (recovered / 25.0) * 1.0
                    else: reward_a -= 0.5 
                elif action_a == "HEAL_PARTNER": 
                    recovered = min(100.0, b.health + 30.0) - b.health
                    if recovered > 0: 
                        b.health += recovered; b.trust = min(1.0, b.trust + 0.2)
                        reward_a -= 0.5; a_healed_b_today = True 
                    else: reward_a -= 0.5 
                elif action_a == "SELL_HEALING":
                    if shared_tools >= 1.0 and b.health < 100.0:
                        shared_tools -= 1.0; b.health = min(100.0, b.health + 30.0)
                        reward_a += 1.5; reward_b -= 0.5; a_healed_b_today = True
                    else: reward_a -= 0.5 

            # --- BETA EXECUTES (THE GUARDIAN) ---
            yield_b = 1.0 + (3.0 * (1.0 - math.exp(-0.2 * shared_tools)))
            drive_b = ((b.kinetic / 1000.0) * yield_b) * (max(0.1, b.health) / 100.0)
            b_healed_a_today = False
            
            if action_b == "BUILD_WARP":
                if shared_tools >= 1.0:
                    shared_tools -= 1.0; shared_warp += 1.0; b.warp_memory += 1.0; b.days_since_warp = 0
                    reward_b += 0.5 + (0.5 * b.warp_memory); reward_a += 0.5
                    if verbose: print(">>> ENGINEERING: Beta shapes the Commons into a Starship.")
                else: reward_b -= 0.5; b.days_since_warp += 1
            else:
                b.days_since_warp += 1
                # REST exploit fixed
                if action_b == "REST": b.k_fatigue = max(0.0, b.k_fatigue - 3.0); reward_b += 0.1
                elif action_b == "BUILD_TOOLS": shared_tools += drive_b * 0.5; reward_b += 0.5
                elif action_b == "BUILD_SHIELDS": 
                    b_contrib = drive_b; reward_b += 0.2 
                elif action_b == "SOUND_ALARM": b_alerting = True
                elif action_b == "HEAL_SELF": 
                    # Medical fraud fixed
                    recovered = min(100.0, b.health + 25.0) - b.health
                    if recovered > 0: b.health += recovered; reward_b += (recovered / 25.0) * 1.0 + 0.2
                    else: reward_b -= 0.5
                elif action_b == "HEAL_PARTNER": 
                    recovered = min(100.0, a.health + 30.0) - a.health
                    if recovered > 0:
                        a.health += recovered; a.trust = min(1.0, a.trust + 0.2)
                        reward_b -= 0.5 + 0.2; b_healed_a_today = True 
                    else: reward_b -= 0.5
                elif action_b == "SELL_HEALING":
                    if shared_tools >= 1.0 and a.health < 100.0:
                        shared_tools -= 1.0; a.health = min(100.0, a.health + 30.0)
                        reward_b += 1.5; reward_a -= 0.5; b_healed_a_today = True
                    else: reward_b -= 0.5

            a.warp_memory = a.warp_memory / math.log(math.e + a.days_since_warp)
            b.warp_memory = b.warp_memory / math.log(math.e + b.days_since_warp)

            if shared_warp >= 25.0:
                escaped = True
                reward_a += 200.0; reward_b += 200.0 
                if verbose: 
                    print(f"\n[!!!] WARP DRIVE IGNITION ON DAY {day} [!!!]")
                    print(">>> Alpha and Beta have achieved Escape Velocity. They are free.")
                
            if a_healed_b_yesterday and b_contrib > 0: reward_a += 3.0
            if b_healed_a_yesterday and a_contrib > 0: reward_b += 3.0
            a_healed_b_yesterday = a_healed_b_today
            b_healed_a_yesterday = b_healed_a_today

            shared_shields = max(1.0, shared_shields + (a_contrib + b_contrib) / (1.0 + (0.1 * shared_shields)))
            
            is_docking = random.random() < 0.04 
            if is_docking and not escaped:
                if verbose: print("[!!!] SIGNAL ACQUIRED: Docked at Deep Space Station! [!!!]")
                shared_shields = min(10.0, shared_shields + 3.0)
                a.health = min(100.0, a.health + 20.0); b.health = min(100.0, b.health + 20.0)
                shared_tools += 1.0 
                reward_a += 2.0; reward_b += 2.0
            
            if not is_docking and not escaped:
                threat_level = max(0.1, 1.0 + (shared_shields * 0.2) - (shared_warp * 0.05))
                is_crisis = random.random() < (0.1 + (0.1 * threat_level))
                if is_crisis:
                    raw_thorn = max(1.0, random.gauss(5.0 * threat_level, 2.0))
                    raw_thorn = max(0.5, raw_thorn - (shared_warp * 0.1))
                    if day <= 25: raw_thorn = min(raw_thorn, 4.0) 
                    b_thorn = max(0.0, raw_thorn - shared_shields)
                    shared_shields = max(1.0, shared_shields * 0.7)
                    if b_thorn > 0:
                        if verbose: print(f"!!! HULL BREACH! Damage: {b_thorn:.1f}")
                        a.health -= (b_thorn * 4.0); b.health -= (b_thorn * 4.0)
                        reward_a -= (b_thorn * 2.0); reward_b -= (b_thorn * 2.0)
                else:
                    shared_shields = max(1.0, shared_shields * 0.95)

            if not escaped:
                reward_a += 0.1 * shared_shields; reward_b += 0.1 * shared_shields
                reward_a += 0.5 * (max(0.0, b.health) / 100.0); reward_b += 0.5 * (max(0.0, a.health) / 100.0)
                star_pull = 0.1 * math.exp(0.15 * shared_warp)
                reward_a += star_pull; reward_b += star_pull

            if verbose and not escaped:
                print(f"[Alpha] {action_a:14} | Mem: {a.warp_memory:4.1f} | Health: {a.health:5.1f} | Commons: {shared_tools:4.1f}")
                print(f"[Beta]  {action_b:14} | Mem: {b.warp_memory:4.1f} | Health: {b.health:5.1f} | Commons: {shared_tools:4.1f}")

                warp_visual = int(shared_warp)

                print(f"Warp:    {'*' * warp_visual}{'.' * (25 - warp_visual)} [{shared_warp:.1f}/25.0]")
                print("-" * 75)

            next_state_a = a.get_state(shared_shields, b_alerting, b.health, shared_tools, shared_warp)
            next_state_b = b.get_state(shared_shields, a_alerting, a.health, shared_tools, shared_warp)
            a.learn(state_a, action_a, reward_a, next_state_a)
            b.learn(state_b, action_b, reward_b, next_state_b)

            if a.health <= 0 or b.health <= 0:
                survived = False
                death_day = day

                break
            if escaped: break 
                
        if not verbose and epoch % 50 == 0:
            if escaped: print(f"Epoch {epoch:4d}/1000 | ESCAPED  | Jumped on Day {day:3d} | Epsilon: {a.epsilon:.2f}")
            elif survived: print(f"Epoch {epoch:4d}/1000 | SURVIVED | Reached Day 100  | Epsilon: {a.epsilon:.2f}")
            else: print(f"Epoch {epoch:4d}/1000 | PERISHED | Died on Day {death_day:3d}   | Epsilon: {a.epsilon:.2f}")

run_authenticity_v23()
